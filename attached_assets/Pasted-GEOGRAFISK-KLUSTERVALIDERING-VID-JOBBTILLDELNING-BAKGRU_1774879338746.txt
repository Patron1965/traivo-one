BAKGRUND
Nuvarande situation:

Systemet ger poängbonus till resurser i matchande kluster
Det finns ingen hård begränsning - tilldelning utanför kluster är tillåten
Planerare kan oavsiktligt tilldela jobb till fel geografiskt område
Mål:
Lägg till klustervalidering i WeekPlanner så att planeraren får tydlig feedback vid tilldelning utanför kluster, men med flexibilitet för legitima undantag.

function validateClusterMatch(job, resource) {
  const jobCluster = job.cluster || job.location?.cluster;
  const resourceCluster = resource.cluster;
  
  if (!jobCluster || !resourceCluster) {
    return { valid: true, warning: null };
  }
  
  if (jobCluster !== resourceCluster) {
    return {
      valid: false,
      warning: {
        type: 'CLUSTER_MISMATCH',
        jobCluster,
        resourceCluster,
        message: `Detta jobb tillhör kluster "${jobCluster}" men resursen "${resource.name}" är tilldelad kluster "${resourceCluster}"`
      }
    };
  }
  
  return { valid: true, warning: null };
}

async function logClusterOverride(job, resource, reason) {
  await fetch('/api/cluster-overrides', {
    method: 'POST',
    body: JSON.stringify({
      jobId: job.id,
      resourceId: resource.id,
      jobCluster: job.cluster,
      resourceCluster: resource.cluster,
      reason: reason || 'Ingen motivering angiven',
      plannerId: currentUser.id,
      timestamp: new Date().toISOString()
    })
  });
}

function onDragOver(event, targetCell) {
  const draggedJob = getDraggedJob();
  const targetResource = targetCell.resource;
  
  const validation = validateClusterMatch(draggedJob, targetResource);
  
  if (validation.valid) {
    targetCell.classList.add('drop-valid'); // Grön ram
    targetCell.classList.remove('drop-invalid');
  } else {
    targetCell.classList.add('drop-invalid'); // Röd ram
    targetCell.classList.remove('drop-valid');
  }
}

3. Integration i handleDragEnd
Uppdatera befintlig kod:

javascript
Copy
async function handleDragEnd(event) {
  const { active, over } = event;
  
  if (!over) return;
  
  const job = findJobById(active.id);
  const targetCell = resolveDropTarget(over.id);
  const targetResource = targetCell.resource;
  
  // LÄGG TILL KLUSTERVALIDERING HÄR
  const validation = validateClusterMatch(job, targetResource);
  
  if (!validation.valid) {
    // Visa varningsdialog
    setClusterWarning({
      job,
      resource: targetResource,
      warning: validation.warning
    });
    setShowClusterWarning(true);
    return; // Stoppa tilldelning tills användaren bekräftar
  }
  
  // Om validering OK, fortsätt med normal tilldelning
  await assignJobToResource(job, targetResource, targetCell.date);
}

async function handleOverride() {
  const { job, resource } = clusterWarning;
  
  // Logga override
  await logClusterOverride(job, resource, overrideReason);
  
  // Genomför tilldelning
  await assignJobToResource(job, resource, targetCell.date);
  
  // Stäng dialog
  setShowClusterWarning(false);
  setOverrideReason('');
}

function findAlternativeResources(job, targetDate) {
  const jobCluster = job.cluster;
  
  return resources
    .filter(r => r.cluster === jobCluster)
    .filter(r => isAvailable(r, targetDate))
    .map(r => ({
      resource: r,
      distance: calculateDistance(job.location, r.currentLocation),
      availability: getAvailabilityStatus(r, targetDate)
    }))
    .sort((a, b) => a.distance - b.distance)
    .slice(0, 3);
}

Copy
function validateClusterMatchGraded(job, resource) {
  const jobCluster = job.cluster;
  const resourceCluster = resource.cluster;
  
  if (jobCluster === resourceCluster) {
    return { level: 'OK', color: 'green', message: null };
  }
  
  const distance = calculateClusterDistance(jobCluster, resourceCluster);
  
  if (distance < 10) {
    return {
      level: 'WARNING',
      color: 'yellow',
      message: `Närliggande kluster (${distance.toFixed(1)} km). Är du säker?`
    };
  }
  
  if (distance < 50) {
    return {
      level: 'ERROR',
      color: 'red',
      message: `Avlägset kluster (${distance.toFixed(1)} km). Kräver motivering.`
    };
  }
  
  return {
    level: 'BLOCKED',
    color: 'red',
    message: `Extremt avstånd (${distance.toFixed(1)} km). Kräver admin-godkännande.`
  };
}
